cmake_minimum_required(VERSION 3.24)
project(PLPT VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
  message(FATAL_ERROR "PLPT requires LLVM clang++ (C++20). Set CMAKE_CXX_COMPILER=clang++.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "PLPT targets x64 only in this preview.")
endif()
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Supported platforms: Windows x64 first, Linux x64 second.")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|X86_64)$")
  message(FATAL_ERROR "Only x86-64 is qualified for this preview")
endif()
if(WIN32)
  if(NOT CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC")
    message(FATAL_ERROR "Windows builds use clang++ with the Microsoft SDK/STL ABI. See docs/BUILD.md.")
  endif()
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  # Les .manifest ajoutés aux cibles passent par /MANIFESTINPUT: (CMake).
  # Pas de seconde ressource 1/24 qui entrerait en conflit avec /MANIFEST:EMBED.
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(PLPT_BUILD_UI "Build the native SDL3/ImGui monitor (downloads pinned libraries)" OFF)
option(PLPT_HEADLESS_UI "SDL dummy driver build for Linux smoke tests, not a desktop release" OFF)
option(PLPT_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer (Linux)" OFF)
option(PLPT_BUILD_FUZZER "Build a bounded protocol fuzz target (Linux Clang)" OFF)
include(CTest)
find_package(Threads REQUIRED)
add_library(plpt_options INTERFACE)
target_compile_features(plpt_options INTERFACE cxx_std_20)
if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
  message(FATAL_ERROR "Use clang++.exe, not clang-cl.exe (GNU driver, MSVC ABI on Windows).")
endif()
target_compile_options(plpt_options INTERFACE -Wall -Wextra -Wpedantic -Werror -Wshadow)
if(WIN32)
  target_compile_definitions(plpt_options INTERFACE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00)
endif()
if(PLPT_SANITIZE)
  if(WIN32)
    message(FATAL_ERROR "This sanitizer profile is Linux-only; Windows native runtime tests remain required.")
  endif()
  target_compile_options(plpt_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  target_link_options(plpt_options INTERFACE -fsanitize=address,undefined -fno-sanitize-recover=all)
endif()
add_library(plpt_core STATIC
  engine/core/random.cpp engine/diagnostics/audit.cpp
  engine/discovery/process_tree.cpp engine/discovery/terminal_classifier.cpp
  engine/discovery/monitor.cpp engine/service_registry/registry.cpp
  engine/port_allocator/port_allocator.cpp)
if(WIN32)
  target_sources(plpt_core PRIVATE engine/platform/windows/windows_platform.cpp)
  target_link_libraries(plpt_core PUBLIC ws2_32 iphlpapi bcrypt)
else()
  target_sources(plpt_core PRIVATE engine/platform/linux/linux_platform.cpp)
endif()
target_include_directories(plpt_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(plpt_core PUBLIC plpt_options Threads::Threads)
add_executable(plpt-monitor app/host_shell/main.cpp)
target_link_libraries(plpt-monitor PRIVATE plpt_core)
if(WIN32)
  target_sources(plpt-monitor PRIVATE ${PROJECT_SOURCE_DIR}/packaging/windows/app.manifest)
endif()
install(TARGETS plpt-monitor RUNTIME DESTINATION bin)

# Modèles F/G/H : non liés dans l'application ni installés. Pas de handshake simulé.
if(BUILD_TESTING OR PLPT_BUILD_FUZZER)
  add_library(plpt_models STATIC engine/session/session.cpp engine/messaging/messaging.cpp
    engine/protocol/envelope.cpp engine/resource_manager/resource.cpp)
  target_link_libraries(plpt_models PUBLIC plpt_core)
endif()
if(BUILD_TESTING)
  add_executable(plpt-unit-tests tests/unit/test_main.cpp)
  target_link_libraries(plpt-unit-tests PRIVATE plpt_models)
  add_executable(plpt-platform-tests tests/integration/platform_tests.cpp)
  target_link_libraries(plpt-platform-tests PRIVATE plpt_core)
  add_test(NAME unit COMMAND plpt-unit-tests)
  add_test(NAME platform COMMAND plpt-platform-tests)
  add_test(NAME ipv6 COMMAND plpt-platform-tests --ipv6)
  set_tests_properties(ipv6 PROPERTIES SKIP_RETURN_CODE 77)
  add_test(NAME cli-version COMMAND plpt-monitor --version)
  add_test(NAME cli-arguments COMMAND ${CMAKE_COMMAND} -DPLPT_MONITOR=$<TARGET_FILE:plpt-monitor>
    -P ${PROJECT_SOURCE_DIR}/tests/cli_contract.cmake)
  set_tests_properties(unit platform ipv6 cli-version cli-arguments PROPERTIES TIMEOUT 60)
endif()
if(PLPT_BUILD_FUZZER)
  if(WIN32)
    message(FATAL_ERROR "Fuzzer configured for Linux only in this preview")
  endif()
  target_compile_options(plpt_models PRIVATE -fsanitize=fuzzer-no-link)
  add_executable(plpt-fuzz-envelope tests/fuzz/envelope_fuzz.cpp)
  target_link_libraries(plpt-fuzz-envelope PRIVATE plpt_models)
  target_compile_options(plpt-fuzz-envelope PRIVATE -fsanitize=fuzzer)
  target_link_options(plpt-fuzz-envelope PRIVATE -fsanitize=fuzzer)
endif()
if(PLPT_BUILD_UI)
  include(cmake/Desktop.cmake)
endif()
install(FILES README.md DESTINATION share/doc/plpt)
install(DIRECTORY docs/ DESTINATION share/doc/plpt/docs)
install(FILES third_party/DEPENDENCIES.md DESTINATION share/doc/plpt/third_party)
set(CPACK_PACKAGE_NAME "PLPT-local-monitor-preview")
set(CPACK_PACKAGE_VENDOR "PLPT project")
set(CPACK_PACKAGE_VERSION "0.1.0")
set(CPACK_PACKAGE_FILE_NAME "PLPT-0.1.0-preview-${CMAKE_SYSTEM_NAME}-x64")
if(WIN32)
  set(CPACK_GENERATOR ZIP)
else()
  set(CPACK_GENERATOR TGZ)
endif()
include(CPack)
